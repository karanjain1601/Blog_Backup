Temporal-Text Alignment

# MIL-NCE: multiple-instance NCE for noisy narrations\nimport torch\n\ndef mil_nce_loss(vid_feats, txt_feats, K=4, temp=0.07):\n    # vid_feats: (B, D), txt_feats: (B*K, D)\n    sims = vid_feats @ txt_feats.T / temp  # (B, B*K)\n    # Each video matches a bag of K nearby narrations\n    loss = -torch.logsumexp(sims[:, :K], dim=1).mean()\n    return loss

Contrastive Video-Text Learning

# CLIP4Clip video-text retrieval\ndef retrieve_video(video_path, captions, model):\n    frames = sample_frames(video_path, n=8)\n    vid_feat = model.encode_image(frames).mean(0)  # mean pool\n    txt_feats = model.encode_text(captions)\n    scores = vid_feat @ txt_feats.T\n    top_k = scores.topk(5)\n    return [(captions[i], scores[i].item()) for i in top_k.indices]

Zero-Shot Video Understanding

# Zero-shot action recognition with CLIP\nclass_names = ['running', 'cooking', 'swimming', 'playing guitar']\ntxt_feats = clip.encode_text(class_names)     # (C, 512)\n\nframes = sample_frames(video, n=8)\nframe_feats = clip.encode_image(frames)       # (8, 512)\nvid_feat = frame_feats.mean(0, keepdim=True)  # (1, 512)\n\nlogits = vid_feat @ txt_feats.T\npred = class_names[logits.argmax().item()]

tip: Uniform frame sampling (8 frames) is surprisingly competitive with dense sampling for most video-language tasks. Only for fine-grained temporal understanding (action boundary detection) does dense sampling help significantly.
