2D Convolution as Cross-Correlation

import numpy as np
from scipy.ndimage import convolve

# Manual cross-correlation with a 3x3 Sobel-X kernel
sobel_x = np.array([[-1, 0, 1],
                     [-2, 0, 2],
                     [-1, 0, 1]], dtype=np.float32)

img = np.random.randint(0, 256, (8, 8)).astype(np.float32)
manual = np.correlate2d(img, sobel_x, mode='same')
# True convolution flips the kernel before sliding
scipy_result = convolve(img, sobel_x[::-1, ::-1])
print(np.allclose(manual, scipy_result))  # True

Gaussian Blur and Smoothing

import cv2
import numpy as np

img = cv2.imread('image.jpg', cv2.IMREAD_GRAYSCALE)

blur1 = cv2.GaussianBlur(img, (0, 0), sigmaX=1)
blur3 = cv2.GaussianBlur(img, (0, 0), sigmaX=3)
blur5 = cv2.GaussianBlur(img, (0, 0), sigmaX=5)

# Difference of Gaussians approximates LoG (blob detection)
dog = blur1.astype(np.float32) - blur3.astype(np.float32)
print(f'sigma=1 std: {blur1.std():.2f}, sigma=5 std: {blur5.std():.2f}')
print(f'DoG range: [{dog.min():.1f}, {dog.max():.1f}]')

Laplacian and Sharpness

import cv2

def sharpness_score(image_path, threshold=100.0):
    img = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
    lap = cv2.Laplacian(img, cv2.CV_64F)
    score = lap.var()
    is_blurry = score < threshold
    status = 'BLURRY' if is_blurry else 'SHARP'
    print(f'Laplacian variance: {score:.2f}')
    print(f'Image is {status}')
    return score

sharpness_score('photo.jpg')

import numpy as np
from scipy.ndimage import convolve, convolve1d

sigma = 2.0
k = int(6 * sigma + 1) | 1
x = np.arange(-(k // 2), k // 2 + 1)
g1d = np.exp(-x**2 / (2 * sigma**2))
g1d /= g1d.sum()

img = np.random.rand(64, 64)
# Separable: apply 1D kernel along rows then columns
blurred_sep = convolve1d(convolve1d(img, g1d, axis=1), g1d, axis=0)
g2d = np.outer(g1d, g1d)
blurred_2d = convolve(img, g2d)
print(np.allclose(blurred_sep, blurred_2d, atol=1e-10))  # True
