import torch, timm

def train_from_scratch(model_name, dataset, n_samples):
    subset = Subset(dataset, indices=list(range(n_samples)))
    loader = DataLoader(subset, batch_size=128, shuffle=True)

    optimizer = torch.optim.AdamW(model.parameters(), lr=3e-4, weight_decay=0.05)
    criterion = torch.nn.CrossEntropyLoss()
    for epoch in range(100):
        for x, y in loader:

Computational Scaling

import timm, torch, time

def benchmark_throughput(model_names, batch=64, size=(3, 224, 224)):
    x = torch.randn(batch, *size).cuda()
    results = {}
    for name in model_names:
        m = timm.create_model(name, pretrained=False).cuda().eval()
        with torch.no_grad():
            for _ in range(10): m(x)    # warmup
        torch.cuda.synchronize(); t0 = time.perf_counter()
        with torch.no_grad():
            for _ in range(50): m(x)
        torch.cuda.synchronize()
        results[name] = round(batch * 50 / (time.perf_counter() - t0))
    return results  # img/s: compare resnet50, vit_b_16, convnext_base

tip: ViT's global attention is a feature, not a bug — but it requires more data to discover useful structure. For tasks with strong local patterns (medical imaging, texture) and limited data, ConvNeXt often beats ViT-B.

Task-Specific Strengths

import torch

def attention_rollout(model, image):
    attns, hooks = [], []
    for blk in model.blocks:
        hooks.append(blk.attn.register_forward_hook(
            lambda m, i, o: attns.append(o[1].detach())))
    with torch.no_grad(): model(image.unsqueeze(0).cuda())
    [h.remove() for h in hooks]
    n = attns[0].shape[-1]
    result = torch.eye(n).to(attns[0])
    for a in attns:
        a = a.mean(1)[0] + torch.eye(n).to(a)
        result = (a / a.sum(-1, keepdim=True)) @ result
    return result[0, 1:].reshape(int((n - 1) ** 0.5), -1).cpu().numpy()
