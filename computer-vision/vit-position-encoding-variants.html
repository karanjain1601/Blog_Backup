Why Positional Encoding Matters

import torch, torch.nn as nn, torch.nn.functional as F

class LearnedPosEmbed(nn.Module):
    def __init__(self, num_patches, dim):
        super().__init__()
        self.pos_embed = nn.Parameter(torch.zeros(1, num_patches + 1, dim))

    def interpolate_to(self, new_h, new_w, patch_size=16):
        cls, patch = self.pos_embed[:, :1], self.pos_embed[:, 1:]
        gs = int(patch.shape[1] ** 0.5)
        patch = patch.reshape(1, gs, gs, -1).permute(0, 3, 1, 2)
        nh, nw = new_h // patch_size, new_w // patch_size
        patch = F.interpolate(patch, (nh, nw), mode='bicubic', align_corners=False)
        patch = patch.permute(0, 2, 3, 1).reshape(1, nh * nw, -1)
        return torch.cat([cls, patch], dim=1)

import torch

def build_2d_sinusoidal(h_patches, w_patches, dim):
    assert dim % 4 == 0
    d = dim // 4
    inv_freq = 1.0 / (10000 ** (torch.arange(0, d, dtype=torch.float) / d))
    rows = torch.arange(h_patches).float().unsqueeze(1)
    cols = torch.arange(w_patches).float().unsqueeze(1)
    row_enc = torch.cat([torch.sin(rows * inv_freq), torch.cos(rows * inv_freq)], dim=-1)
    col_enc = torch.cat([torch.sin(cols * inv_freq), torch.cos(cols * inv_freq)], dim=-1)
    row_enc = row_enc.unsqueeze(1).expand(-1, w_patches, -1)
    col_enc = col_enc.unsqueeze(0).expand(h_patches, -1, -1)
    return torch.cat([row_enc, col_enc], dim=-1).reshape(-1, dim)

import torch

def apply_2d_rope(q, k, h_patches, w_patches, head_dim):
    half = head_dim // 4
    inv_freq = 1.0 / (10000 ** (torch.arange(0, half, dtype=q.dtype, device=q.device) / half))
    rows = torch.arange(h_patches, device=q.device).float()
    cols = torch.arange(w_patches, device=q.device).float()
    row_grid, col_grid = torch.meshgrid(rows, cols, indexing='ij')
    row_theta = row_grid.reshape(-1, 1) * inv_freq
    col_theta = col_grid.reshape(-1, 1) * inv_freq
    theta = torch.cat([row_theta, col_theta], dim=-1)
    sin, cos = theta.sin()[None, None], theta.cos()[None, None]
    q_rot = q * cos + rotate_half(q) * sin
    k_rot = k * cos + rotate_half(k) * sin
    return q_rot, k_rot

Resolution Extrapolation

info: Interpolating learned position embeddings to higher resolution requires bicubic interpolation on the 2D grid of patch positions — not linear 1D interpolation. Grid interpolation preserves 2D spatial structure.

import torch

def eval_resolution_transfer(model, val_loader_224, val_loader_384):
    results = {}
    for name, loader in [('224px', val_loader_224), ('384px', val_loader_384)]:
        correct, total = 0, 0
        with torch.no_grad():
            for x, y in loader:
                logits = model(x.cuda())
                correct += (logits.argmax(1) == y.cuda()).sum().item()
                total += y.size(0)
        results[name] = correct / total
        print(f'{name}: {results[name]:.4f}')
    return results
