from transformers import get_cosine_schedule_with_warmup

def build_schedule(optimizer, num_warmup_epochs, total_epochs, steps_per_epoch):
    warmup_steps = num_warmup_epochs * steps_per_epoch
    total_steps = total_epochs * steps_per_epoch
    scheduler = get_cosine_schedule_with_warmup(
        optimizer,
        num_warmup_steps=warmup_steps,
        num_training_steps=total_steps,
    )
    return scheduler

import torch
import torch.nn as nn
from torchvision.transforms.v2 import MixUp

mixup = MixUp(alpha=0.2, num_classes=NUM_CLASSES)
criterion = nn.CrossEntropyLoss(label_smoothing=0.1)

def train_step(model, optimizer, x, y):
    x, y = mixup(x, y)        # y becomes soft labels
    logits = model(x)
    loss = criterion(logits, y)
    optimizer.zero_grad()
    loss.backward()
    torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
    optimizer.step()
    return loss.item()

Layer-Wise LR Decay

warning: Fine-tuning ViT-L with a uniform LR for all layers often degrades early-layer features trained during pre-training. Use layer-wise decay (alpha=0.75) to preserve learned representations while adapting the head.

def get_layer_decay_params(model, base_lr, decay=0.75):
    param_groups = []
    num_layers = model.config.num_hidden_layers
    for name, param in model.named_parameters():
        if not param.requires_grad:
            continue
        if 'embeddings' in name or 'patch_embed' in name:
            layer_idx = 0
        elif 'layer' in name or 'block' in name:
            layer_idx = int(name.split('.')[2]) + 1
        else:
            layer_idx = num_layers + 1  # classification head
        lr = base_lr * (decay ** (num_layers + 1 - layer_idx))
        param_groups.append({'params': [param], 'lr': lr})
    return param_groups

Resolution Change and Patch Adaptation

import torch, torch.nn.functional as F

def resize_pos_embed(model, new_h, new_w, patch_size=16):
    old = model.pos_embed          # (1, N+1, D)
    cls_tok = old[:, :1, :]
    patches = old[:, 1:, :]
    gs_old = int(patches.shape[1] ** 0.5)
    patches = patches.reshape(1, gs_old, gs_old, -1).permute(0, 3, 1, 2)
    nh, nw = new_h // patch_size, new_w // patch_size
    patches = F.interpolate(patches, (nh, nw), mode='bicubic', align_corners=False)
    patches = patches.permute(0, 2, 3, 1).reshape(1, nh * nw, old.shape[-1])
    new_embed = torch.cat([cls_tok, patches], dim=1)
    model.pos_embed = torch.nn.Parameter(new_embed)
